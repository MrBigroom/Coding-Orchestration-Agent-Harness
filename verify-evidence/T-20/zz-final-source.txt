## src/orchestrator/services/git_service.py:221-241  parity_lock
  221| def parity_lock(project_slug: str) -> Iterator[None]:
  222|     """Hold the doctor parity lock for the lifetime of a parity run.
  223| 
  224|     An interrupted run releases the lock with the process, so a later sweep can tell an orphaned
  225|     tree (lock free) from one another process is still using (lock busy). The lock path lives
  226|     beside the managed marker, never inside the throwaway worktree, because the whole worktree is
  227|     deleted by cleanup.
  228|     """
  229|     import fcntl  # POSIX-only, imported here so the module itself stays platform-portable
  230| 
  231|     path = parity_lock_path(project_slug)
  232|     path.parent.mkdir(parents=True, exist_ok=True)
  233|     with path.open("a+") as fh:
  234|         try:
  235|             fcntl.flock(fh, fcntl.LOCK_EX | fcntl.LOCK_NB)
  236|         except BlockingIOError as exc:
  237|             raise ParityBusy(f"another parity run holds {path}") from exc
  238|         try:
  239|             yield
  240|         finally:
  241|             fcntl.flock(fh, fcntl.LOCK_UN)

  462| def _env_variable_names(source: Path) -> list[str]:
  463|     """The SORTED variable NAMES declared in an env file; values are never parsed.
  464| 
  465|     The parity probe must reproduce the operator's *configuration surface* without ever writing a
  466|     credential value into a harness-managed directory (T-20 DoD 1). Only the name before the first
  467|     `=` leaves this function; comments, blank lines and `export` prefixes are handled. The order is
  468|     sorted, matching `sed ... | sort -u` in scripts/orch_env_parity.sh, so the two paths write the
  469|     same name sequence for the same file (T-20 review round 3); first-seen order would silently
  470|     differ whenever the operator's file is not alphabetical.
  471|     """
  472|     names: set[str] = set()
  473|     for raw in source.read_text(encoding="utf-8", errors="replace").splitlines():
  474|         line = raw.strip()
  475|         if not line or line.startswith("#"):
  476|             continue
  477|         if line.startswith("export "):
  478|             line = line[len("export ") :].lstrip()
  479|         name, sep, _value = line.partition("=")
  480|         name = name.strip()
  481|         if not sep or not _ENV_NAME_RE.match(name):
  482|             continue
  483|         names.add(name)
  484|     return sorted(names)
