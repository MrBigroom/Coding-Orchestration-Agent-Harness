  648 | def test_gc_only_deletes_files_render_file_could_have_written(session):
  649 |     """Hand-authored markdown is operator input (docs/memory.md, `import --markdown`).
  650 | 
  651 |     The GC predicate is `in_layout_dir AND not canonical_file AND not owned AND
  652 |     _renderer_output`, and the last term is what protects a human's file: the first line
  653 |     must be exactly `# <stem>` and every later line must be a `- … <!-- mem:N -->` marker.
  654 |     """
  655 |     root = m.memory_root()
  656 |     proj = root / "projects" / "p"
  657 |     proj.mkdir(parents=True, exist_ok=True)
  658 |     hand = proj / "notes.md"
  659 |     hand.write_text("# Hand-written notes\n\n- a line someone typed by hand\n")
  660 |     hand_title_only = proj / "scratch.md"  # right heading shape, prose body -> kept
  661 |     hand_title_only.write_text("# scratch\n\nremember to check the mount path\n")
  662 |     header = proj / "conventions.md"  # canonical name holding an ensure_layout header
  663 |     header.write_text("# Project p — conventions\n")
  664 |     ghost = proj / "p.md"  # render_file's own output, now with no owning entry
  665 |     ghost.write_text("# p\n")
  666 |     stale = proj / "old-scope.md"  # rendered with a marker whose entry has moved away
  667 |     stale.write_text("# old-scope\n\n- dead line  <!-- mem:999 -->\n")
  668 |     outside = root.parent / "zz-p.md"  # not in a layout directory
  669 |     outside.write_text("# p\n")
  670 | 
  671 |     removed = m.gc_orphan_files(session)
  672 |     assert removed == [stale, ghost]  # sorted by path, and only the two renderer-shaped files
  673 |     assert hand.exists() and hand_title_only.exists() and header.exists()
  674 |     assert not ghost.exists() and not stale.exists()
  675 |     assert outside.exists()
