+
+    _ensure_ready()
+    with _session() as s:
+        if gc_only:

+            repair = m.ScopeRepair(removed_files=removed or listed, dry_run=dry_run)
+        else:
+            repair = m.fix_scopes(s, dry_run=dry_run, gc=not no_gc)
+    verb = "would move" if dry_run else "moved"
