def handoff_snapshot(path: Path) -> dict[str, Any] | None:
    """A cheap fingerprint of the handoff file as it stands right now.

    Used to answer one question at the end of a run: is the file still the exact leftover that was
    already in the worktree when the round started? A stat-only check is not enough (a same-size
    rewrite with a coarse timestamp would collide), so the digest is included. An unreadable file
    still gets a stat-based fingerprint; ingestion will report why it cannot be used.
    """
    try:
        st = path.stat()
    except OSError:
        return None
    if not path.is_file():
        return None
    try:
        digest = hashlib.sha256(path.read_bytes()).hexdigest()
    except OSError:
        digest = None
    return {"mtime_ns": st.st_mtime_ns, "size": st.st_size, "sha256": digest}

def read_handoff_file(path: Path) -> tuple[Handoff | None, str | None]:
    """Read a worktree `handoff.json`. Returns `(handoff, error)`.

    `(None, None)` means "no file" — there is nothing to salvage. `(None, reason)` means a file
    exists but cannot be used: that is a logged fact, never an exception, because a truncated
    write from a killed process is the normal shape of this case.
    """
    if not path.is_file():
        return None, None
    try:
        raw = path.read_text(encoding="utf-8", errors="replace")
    except OSError as e:
        return None, f"{path} could not be read: {e}"
    if not raw.strip():
        return None, f"{path} is empty (0 usable bytes)"
    try:
        data = json.loads(raw)
    except json.JSONDecodeError as e:
        return None, f"{path} is not valid JSON ({e}); {len(raw)} bytes on disk"
    try:
        return Handoff.model_validate(data), None
    except ValueError as e:
        return None, f"{path} is valid JSON but does not validate as Handoff: {_one_line(str(e))}"

def delivery_evidence(
    wt: Path,
    base_branch: str,
    handoff: Handoff | None,
    *,
    arrival: str,
    exit_state: str,
    uncommitted_before_sweep: list[str] | None = None,
) -> dict[str, Any]:
    """What git says about a handoff's claims, computed at ingestion time (T-10 DoD 7).

    A handoff is a *statement*. `git log base..HEAD`, the diffstat and `git status --porcelain
    -uall` are evidence. The measured case this exists for (run 13, 2026-11-04): a cancelled run
    whose 1164-byte handoff claimed four code/doc fixes while it had made zero commits and the
    worktree had no tracked modifications — and the old "claimed files exist on disk" check passed,
    because the files did exist as uncommitted scratch.

    Each claim is classified as exactly one of `artifact_service.CLAIM_STATUSES`, first match
    wins: `worker_commit` (delivered by a Worker-authored commit on this branch),
    `recovery_commit_only` (only inside a harness recovery commit — wreckage no Worker
    self-checked), `uncommitted_only` (a promise, still on the worktree, not on the branch),
    `absent` (nowhere). The structured flag `claims_all_backed_by_worker_commits` is the
    one-line answer a reviewer can act on.
    """
    commits = git_service.commits_against_base(wt, base_branch)
    worker_paths: set[str] = set()
    recovery_paths: set[str] = set()
    for c in commits:
        paths = {_normalize_rel(f["path"]) for f in c.files}
        (recovery_paths if c.recovery else worker_paths).update(paths)
    split = git_service.status_split(wt)
    uncommitted = {_normalize_rel(p) for p in (*split.tracked_modified, *split.untracked_added)}
    claims = [_normalize_rel(f) for f in (handoff.files_changed if handoff else [])]
    status: dict[str, str] = {}
    for claim in claims:
        if claim in worker_paths:
            status[claim] = artifact_service.CLAIM_WORKER_COMMIT
        elif claim in recovery_paths:
            status[claim] = artifact_service.CLAIM_RECOVERY_COMMIT
        elif claim in uncommitted:
            status[claim] = artifact_service.CLAIM_UNCOMMITTED
        else:
            status[claim] = artifact_service.CLAIM_ABSENT
    counts: dict[str, int] = {}
    for s in status.values():
        counts[s] = counts.get(s, 0) + 1
    _, diffstat = git_service.diff_against_base(wt, base_branch)
    return {
        "arrival": arrival,
        "exit_state": exit_state,
        "base_branch": base_branch,
        "commits": [c.as_dict() for c in commits],
        "commit_count": len(commits),
        "worker_commit_count": sum(1 for c in commits if not c.recovery),
        "recovery_commit_count": sum(1 for c in commits if c.recovery),
        "diffstat": diffstat,
        "porcelain_at_ingestion": [
            line for line in status_porcelain(wt).splitlines() if line.strip()
        ],
        "uncommitted_before_harness_sweep": sorted(uncommitted_before_sweep or []),
        "claimed_files": sorted(claims),
        "claimed_count": len(claims),
        "claim_status": status,
        "claim_counts": counts,
        "claims_all_backed_by_worker_commits": bool(claims)
        and all(s == artifact_service.CLAIM_WORKER_COMMIT for s in status.values()),
    }

def try_salvage_handoff(
    session: Session,
    task: Task,
    run: Run,
    project: Project,
    wt: Path,
    handoff_path: Path,
    harness_name: str | None,
    result: AdapterResult,
    uncommitted_before_sweep: list[str],
    stale_snapshot: dict[str, Any] | None = None,
) -> Handoff | None:
    """Ingest a handoff the killed Worker already wrote (T-10 DoD 1 + 2).

    Returns the salvaged Handoff when the worktree holds one that validates, else None — and None
    means "behave exactly as before this task existed": `RUN_TIMEOUT`/`RUN_FAILED` on the task and
    the task back to `planned`. A present-but-unusable file logs `handoff_salvage_failed` instead of
    crashing the run, because a half-written file is the expected shape of a killed process.
    """
    handoff, error = read_handoff_file(handoff_path)
    if stale_snapshot is not None and handoff_snapshot(handoff_path) == stale_snapshot:
        # The round-start archive failed, so the previous round's file was left in place — and the
        # Worker never replaced it. Ingesting it here would attribute a previous round's
        # submission to this run, which is exactly the provenance failure the salvage path exists
        # to avoid. Fail closed: this round produced nothing and is reported as incomplete.
        log_service.log_event(
            session,
            "handoff_salvage_failed",
            structured_log.attach(
                f"{run.short_id} ({run.status}) left a handoff.json in the worktree, but it is "
                "byte-for-byte unchanged since the round started: it is the previous round's "
                "leftover (the round-start archive could not copy it), not this run's submission. "
                "Refusing to salvage it and falling back to the normal non-clean-exit path.",
                _salvage_log_payload(task, run, result, None, SALVAGE_MODE_NON_CLEAN_EXIT, None)
                | {"reason": "unchanged since round start", "stale": True},
            ),
            role="supervisor",
            task_id=task.id,
            run_id=run.id,
        )
        return None
    if handoff is None and error is None:
        return None
    if handoff is None:
        log_service.log_event(
            session,
            "handoff_salvage_failed",
            structured_log.attach(
                f"{run.short_id} ({run.status}) left {handoff_path} in the worktree but it could "
                f"not be ingested; falling back to the {run.status} path: {error}",
                _salvage_log_payload(task, run, result, None, SALVAGE_MODE_NON_CLEAN_EXIT, None)
                | {"reason": error},
            ),
            role="supervisor",
            task_id=task.id,
            run_id=run.id,
        )
        return None
    evidence = delivery_evidence(
        wt,
        project.base_branch,
        handoff,
        arrival=artifact_service.HANDOFF_ARRIVAL_SALVAGED,
        exit_state=exit_state_text(run, result),
        uncommitted_before_sweep=uncommitted_before_sweep,
    )
    archive_handoff_artifacts(run, handoff)
    _write_run_json(
        run,
        task,
        harness_name,
        result,
        arrival=artifact_service.HANDOFF_ARRIVAL_SALVAGED,
        evidence=evidence,
    )
    if handoff.memory_suggestions:
        # Pending, never auto-approved: a suggestion from a killed run is no more self-certifying
        # than the handoff it came with, and the Supervisor still reviews it.
        memory_service.suggest_from_handoff(
            session,
            handoff.memory_suggestions,
            source=f"salvaged-handoff:{run.short_id}",
            project_slug=project.slug,
            role="worker",
        )
    log_service.log_event(
        session,
        "handoff_salvaged",
        structured_log.attach(
            f"{run.short_id} was killed ({run.status}) after the Worker had already written a "
            f"valid handoff.json to the worktree, out of band: the adapter never returned it on a "
            f"clean exit, so the old code path discarded it. Salvaged into {run.artifacts_path} as "
            f"handoff.json + handoff.md; run.status stays {run.status} and the Supervisor gate "
            f"(not the Worker's own claim) decides the round. Claims backed only by uncommitted "
            f"worktree state: "
            f"{'yes' if not evidence['claims_all_backed_by_worker_commits'] else 'none'}.",
            _salvage_log_payload(
                task, run, result, handoff, SALVAGE_MODE_NON_CLEAN_EXIT, evidence
            ),
        ),
        role="supervisor",
        task_id=task.id,
        run_id=run.id,
        commit_sha=run.commit_sha,
    )
    return handoff

def archive_before_round_start(
    session: Session,
    task: Task,
    run: Run,
    project: Project,
    handoff_path: Path,
) -> bool:
    """Never destroy a pre-existing handoff.json when a round starts (T-10 DoD 3).

    The round-start `unlink(missing_ok=True)` was the second half of the bug: a retry after a
    timeout actively deleted the previous round's submission, so the operator had to copy the file
    out in the gap. Returns True when the caller may unlink — i.e. the submission is provably
    archived (already in `runs/<prev>/handoff.json`, or copied to ORCHESTRATOR_HOME/evidence/).
    On any archive failure it returns False and the file is left alone: keeping a stale handoff in
    the scratch dir is recoverable, losing one is not.
    """
    if not handoff_path.is_file():
        return True
    try:
        raw = handoff_path.read_text(encoding="utf-8", errors="replace")
    except OSError as e:
        log_service.log_event(
            session,
            "handoff_salvage_failed",
            structured_log.attach(
                f"{handoff_path} exists but could not be read before the round-start unlink; "
                "leaving it in place (the unlink was skipped so nothing is destroyed)",
                {"task_id": task.id, "run_id": run.id, "reason": str(e)},
            ),
            role="supervisor",
            task_id=task.id,
            run_id=run.id,
        )
        return False
    handoff, error = read_handoff_file(handoff_path)
    prev = _previous_round_run(session, task, run)
    if handoff is not None and prev is not None and prev.id is not None:
        archived = artifact_service.read_json(prev.id, "handoff.json")
        if archived == handoff.model_dump(mode="json"):
            # SALVAGE_MODE_ALREADY_ARCHIVED: already archived byte-for-byte in that run's
            # artifacts, so nothing is at risk and nothing is copied. Deliberately not an event —
            # an event here would fire on every ordinary retry and mean nothing.
            return True
    key = (
        f"{project.slug}-{task.short_id}-"
        f"{prev.short_id if prev else 'no-previous-run'}-"
        f"{utcnow().strftime('%Y%m%dT%H%M%S%f')}"
    )
    try:
        dest = artifact_service.evidence_dir(EVIDENCE_KIND_PRE_ROUND, key)
        (dest / "handoff.json").write_text(raw, encoding="utf-8")
        if handoff is not None:
            (dest / "handoff.md").write_text(
                artifact_service.render_handoff_md(handoff), encoding="utf-8"
            )
    except OSError as e:
        log_service.log_event(
            session,
            "handoff_salvage_failed",
            structured_log.attach(
                f"could not archive {handoff_path} to the evidence root before the round-start "
                "unlink; leaving it in place (the unlink was skipped so nothing is destroyed)",
                {"task_id": task.id, "run_id": run.id, "reason": str(e)},
            ),
            role="supervisor",
            task_id=task.id,
            run_id=run.id,
        )
        return False
    log_service.log_event(
        session,
        "handoff_salvaged",
        structured_log.attach(
            f"a handoff.json from the previous round was still in the worktree at the start of "
            f"{run.short_id}; archived to {dest} before the round-start unlink "
            f"(source run: {prev.short_id + ' (' + prev.status + ')' if prev else 'unknown'}; "
            f"{'valid handoff' if handoff else 'unusable: ' + (error or 'unknown')}), "
            "so a retry can no longer destroy a submission",
            {
                "mode": SALVAGE_MODE_PRE_ROUND_ARCHIVE,
                "task_id": task.id,
                "run_id": run.id,
                "source_run": prev.short_id if prev else None,
                "source_run_status": prev.status if prev else None,
                "archived_to": str(dest),
                "valid": handoff is not None,
                "parse_error": error,
                "bytes": len(raw),
            },
        ),
        role="supervisor",
        task_id=task.id,
        run_id=run.id,
    )
    return True

async def _gate_salvaged(
    session: Session, task: Task, run: Run, project: Project, handoff: Handoff
) -> dict[str, Any]:
    """Continue to the normal verification + review path carrying a salvaged submission.

    The run's status is already terminal (`timeout`/`failed`/`cancelled`) and the frozen run
    vocabulary has no transition out of a terminal state, so the gate runs WITHOUT moving the run —
    deliberately not `testing`/`completed`: the process really was killed, and pretending otherwise
    would widen the frozen vocabulary and hide the fact that the Worker never saw the gate result.
    `task.handoff_status` becomes `SALVAGED` (distinct from the Worker's own status, from
    `SYNTHESIZED` and from `RUN_*`); see docs/build-loop.md, "Salvaging a handoff from a
    killed run".
    """
    assert run.id is not None
    task.handoff_status = artifact_service.HANDOFF_STATUS_SALVAGED
    session.add(task)
    session.commit()
    if handoff.status == "BLOCKED":
        task_service.block(
            session,
            task,
            f"worker emitted BLOCKED handoff (salvaged from a {run.status} run, so the Worker "
            "never saw this exit): " + "; ".join(handoff.blockers),
            role="supervisor",
        )
        return {
            "run_id": run.id,
            "status": run.status,
            "error": run.error,
            "salvaged": True,
            "decision": "blocked",
            "blockers": handoff.blockers,
        }
    task_service.transition(
        session,
        task,
        TaskStatus.review,
        f"salvaged handoff from a {run.status} run; the Supervisor gate decides on evidence",
        role="supervisor",
    )
    outcome = await supervisor_service.after_run(session, task, run, project, handoff, False)
    return {"run_id": run.id, "status": run.status, "error": run.error, "salvaged": True, **outcome}

                session,
                run,
                RunStatus.completed,
                finished_at=finished,
                duration_ms=int((finished - started).total_seconds() * 1000),
            )
    return {"run_id": run.id, "status": run.status, **outcome}

async def verify(
    session: Session,
    task: Task,
    run: Run,
    project: Project | None,
    handoff: Handoff | None,
    handoff_synthesized: bool,

    # 1. Handoff present
    if handoff is None or handoff_synthesized:
        checks.append(
            VerificationCheck(
                name="handoff present",
                status="fail",
                evidence="worker did not produce a valid handoff.json",
            )
        )
    else:
        checks.append(
            VerificationCheck(
                name="handoff present", status="pass", evidence=f"status {handoff.status}"
            )
        )

    # 1b. Provenance of the submission itself (T-10). Read back from the archived `run.json`, not
    # re-derived, so the gate, the reviewer payload and a human quoting the artifacts agree. An
    # salvaged handoff is NOT worker self-certification: the run was killed, and the gate below is
    # what decides. Informational — it never approves and never rejects by itself.
    arrival = artifact_service.handoff_arrival(run.id)
    if arrival == artifact_service.HANDOFF_ARRIVAL_SALVAGED:
        evidence0 = artifact_service.handoff_delivery_evidence(run.id) or {}
        checks.append(

                name=SALVAGE_CHECK,
                status="pass",
                evidence=(
                    f"run.status={run.status} is unchanged; the submission was read off the "
                    "worktree, not returned by the adapter on a clean exit"
                ),
                notes=(
                    f"{evidence0.get('exit_state') or 'exit state unknown'}. The Worker process "
                    "was killed before it could submit, so this handoff is not a self-certified "
                    "completed run: the deterministic checks below and the Architect review decide "
                    "the round, and a green gate routes it to a human instead of auto-accepting it "
                    "(docs/build-loop.md, 'Salvaging a handoff from a killed run')."
                ),
            )
        )

    # 1c. Claims vs. commits (T-10 DoD 7) is deliberately NOT a verification check. The evidence
    # is computed at ingestion, archived in `run.json` and rendered into the reviewer payload; the
    # reviewer and the operator judge it. A claim that exists only as uncommitted worktree state is
    # a promise, and a harness recovery commit is content the Worker never self-checked — both are
    # labelled in that evidence, but neither may become a new automatic gate here: T-8 owns the
    # recovery-commit machinery and DoD 7 asks for provenance, not extra rejection.

    # 2. Claimed files exist on disk

        claimed_not_changed = sorted(set(handoff.files_changed) - actual) if project else []
        if missing or claimed_not_changed:
            contradicts = True
        checks.append(
            VerificationCheck(
                name="claimed files exist on disk",
                status="fail" if missing else "pass",
                evidence=f"{len(handoff.files_changed)} claimed, {len(missing)} missing",
                notes="; ".join(
                    filter(
                        None,
                        [
                            f"missing: {missing}" if missing else "",
                            f"claimed but not in diff: {claimed_not_changed}"
                            if claimed_not_changed
                            else "",
                            f"changed but not claimed: {unclaimed}" if unclaimed else "",

                    )
                ),
            )
        )
    elif task.kind in (TaskKind.coding, TaskKind.maintenance):
        checks.append(
            VerificationCheck(
                name="claimed files exist on disk",
                status="skipped",
                notes="no files claimed in handoff",
            )
        )

            status="pass" if task.kind == TaskKind.agent and handoff else "not_applicable",

            if task.kind == TaskKind.agent

    )

    # Supervisor side on purpose: a Worker or reviewer that merely *claims* a contract change was
    # authorized cannot make the claim true — the file list and the stored DoD decide it. Worker
    # tasks only: an `agent`/`review` task answers questions and writes no code, and an
    # Architect decomposition run is *supposed* to write docs/plan.md.
    if task.kind in (TaskKind.coding, TaskKind.maintenance):

        recovery_paths = {f["path"] for c in recovery for f in c.files}
        findings = project_service.contract_findings(task, project, cwd)
        if findings["unauthorized"]:
            from_recovery = sorted(set(findings["unauthorized"]) & recovery_paths)
            checks.append(
                VerificationCheck(
                    name=CONTRACT_CHECK,
                    status="fail",
                    evidence=f"{len(findings['unauthorized'])} contract file(s) changed",
                    notes=(
                        "changed without a Definition-of-Done item or acceptance criterion naming "
                        "them: " + ", ".join(findings["unauthorized"]) + ". "
                        + (
                            f"{', '.join(from_recovery)} arrived inside a recovery commit, so it "
                            "is content the Worker never self-checked. "
                            if from_recovery
                            else ""
                        )
                        + "Per docs/build-loop.md this phase may not proceed on its own: the "
                        "finding goes to the human (contract-change request flow)."
                    ),
                )
            )
            log_service.log_event(
                session,
                "contract_change_detected",
                structured_log.attach(
                    "unauthorized contract file change(s) in this run",
                    {
                        "unauthorized": findings["unauthorized"],
                        "authorized": findings["authorized"],
                        "contract_files": project_service.contract_files(),
                        "recovery_commit": bool(from_recovery),
                        "from_recovery_commit": from_recovery,
                        "recovery_commits": [c.short for c in recovery],
                        "task_id": task.id,
                        "run_id": run.id,
                    },
                ),
                role="supervisor",
                task_id=task.id,
                run_id=run.id,
                phase=task_service.phase_label(task),
                commit_sha=head_sha(cwd) if cwd else None,
            )
        elif findings["changed"] or project:
            checks.append(
                VerificationCheck(
                    name=CONTRACT_CHECK,
                    status="pass",
                    evidence=(
                        f"{len(findings['changed'])} contract file(s) changed, all authorized"

            )

        checks.append(
            VerificationCheck(
                name=RECOVERY_CHECK,
                # Informational: a recovery commit is not itself a rejection reason. It always
                # reaches the reviewer (verification.md is part of the review payload) so nothing
                # is quietly folded into "the Worker's own work".
                status="pass",

                notes=(
                    git_service.format_recovery_commit_list(recovery)

                    if recovery

    # failing check like any other, not a footnote) — while `after_run` still routes it to the

    failed = [c for c in checks if c.status == "fail"]
    final = "fail" if failed or contradicts else "pass"

            verifier_kind="supervisor",
            status=final,
            checks_json=[c.model_dump() for c in checks],

            contradicts_worker=contradicts,
        )
    )

        task_id=task.id,
        run_id=run.id,

async def after_run(
    session: Session,
    task: Task,
    run: Run,
    project: Project | None,
    handoff: Handoff | None,
    handoff_synthesized: bool,
) -> dict:

    verification = await verify(session, task, run, project, handoff, handoff_synthesized)
    tests_green = all(c.status != "fail" for c in verification.checks if c.name == "tests")
    outcome: dict = {"verification": verification.final_result}

    if not tests_green or handoff is None:
        _requeue_or_block(session, task, "tests not green or handoff missing; review not triggered")
        outcome["decision"] = task.status
        return outcome

    # Checked before the reviewer runs: a contract edit is a human decision, and letting a model
    # review it first would let an APPROVED verdict wave the finding through.
    held = contract_change_hold(session, task, run, project)
    if held:
        outcome["decision"] = task.status
        outcome["contract_change"] = held
        return outcome

    needs_architect = (
        task.kind in (TaskKind.coding, TaskKind.maintenance, TaskKind.review)
        or task.risk == TaskRisk.high
    )
    review: Review | None = None
    if needs_architect:
        review = await review_service.architect_review(
            session, task, run, project, handoff_synthesized
        )

        if review.verdict == ReviewVerdict.NEEDS_HUMAN:
            task_service.transition(
                session,
                task,
                TaskStatus.blocked,
                "architect verdict NEEDS_HUMAN",
                role="supervisor",
            )
            outcome["decision"] = task.status
            return outcome
        if review.verdict == ReviewVerdict.CHANGES_REQUESTED:
            _requeue_or_block(session, task, "architect requested changes")
            outcome["decision"] = task.status
            return outcome

    if verification.final_result != "pass":
        _requeue_or_block(session, task, "supervisor verification failed (believe the check)")
        outcome["decision"] = task.status
        return outcome

    # Salvaged submission, everything else green. Decision (T-10 DoD 4): `handoff_status=SALVAGED`
    # and the round routes to NEEDS_HUMAN instead of `accepted`. Rationale: the Worker process was
    # killed, so it never saw this gate result and cannot apply reviewer feedback on the same
    # branch, and an early handoff is partly a statement of intent (T-10 DoD 7 measures that). The
    # submission is not lost and the operator is not told the round failed — the round waits for a
    # decision, with handoff/review/verification artifacts attached, and `orchestrator task
    # accept` / the GUI accept button takes it from there (blocked -> review -> accepted, the same
    # path an operator already uses for a contract-change hold). It is never auto-merged.
    if artifact_service.handoff_arrival(run.id) == artifact_service.HANDOFF_ARRIVAL_SALVAGED:
        question = (
            f"salvaged submission from run {run.short_id} ({run.status}): the Worker wrote a valid "
            "handoff.json before the process was killed. Verification passed and the Architect "
            f"review is on file ({review.verdict if review else 'n/a'}), but a killed run cannot "
            "react to feedback and an early handoff can promise more than it delivers — decide "
            "whether this round counts (accept) or goes back to the Worker (retry)."
        )
        task.review_status = ReviewVerdict.NEEDS_HUMAN
        session.add(task)
        session.commit()
        log_service.log_event(
            session,
            "verdict_recorded",
            f"NEEDS_HUMAN {question}",
            role="supervisor",
            task_id=task.id,
            run_id=run.id,
        )
        task_service.transition(session, task, TaskStatus.blocked, question, role="supervisor")
        outcome["decision"] = task.status
        outcome["salvaged"] = True
        return outcome

    outcome |= accept(session, task, run, project, review, verification)
    return outcome

def render_delivery_evidence_md(evidence: dict[str, Any], arrival: str | None = None) -> str:
    """The reviewer-payload rendering of `run.json.handoff_delivery_evidence` (T-10 DoD 7).

    Rendered from the SAME archived numbers the run artifacts carry, so a claim that exists only
    as uncommitted worktree state — or only inside a harness recovery commit — reads as a promise
    and not as delivered work.
    """
    if not evidence:
        return ""
    commits = evidence.get("commits") or []
    status = evidence.get("claim_status") or {}
    counts = evidence.get("claim_counts") or {}
    lines = [f"handoff arrival: {arrival or evidence.get('arrival') or 'unknown'}"]
    if evidence.get("exit_state"):
        lines.append(f"run exit state: {evidence['exit_state']}")
    lines.append(
        f"commits on this branch since {evidence.get('base_branch', 'the base branch')}: "
        f"{evidence.get('commit_count', len(commits))} "
        f"(worker-authored {evidence.get('worker_commit_count', 0)}, "
        f"harness recovery {evidence.get('recovery_commit_count', 0)})"
    )
    for c in commits[:MAX_COMMITS_IN_EVIDENCE]:
        kind = "harness recovery commit, NEVER self-checked" if c.get("recovery") else "commit"
        paths = ", ".join(f"{f['status']} {f['path']}" for f in (c.get("files") or [])[:12])
        lines.append(f"  - {c.get('short')} [{kind}] {c.get('subject')} ({paths or 'no files'})")
    if len(commits) > MAX_COMMITS_IN_EVIDENCE:
        lines.append(f"  - (+{len(commits) - MAX_COMMITS_IN_EVIDENCE} more commits)")
    lines.append(f"diffstat vs merge-base:\n{evidence.get('diffstat') or '  (empty diff)'}")
    dirt = evidence.get("porcelain_at_ingestion") or []
    lines.append(
        "git status --porcelain -uall at ingestion: "
        + ("clean" if not dirt else "\n" + "\n".join(f"  {d}" for d in dirt))
    )
    swept = evidence.get("uncommitted_before_harness_sweep") or []
    lines.append(
        "uncommitted when the run ended (before the harness recovery sweep): "
        + (", ".join(swept) if swept else "none")
    )
    lines.append(
        "claims in handoff.files_changed "
        f"({evidence.get('claimed_count', len(status))}): "
        + (", ".join(f"{k}={v}" for k, v in sorted(counts.items())) or "no files claimed")
    )
    for path, st in sorted(status.items()):
        lines.append(f"  - {path} -> {st}")
    lines.append(
        "structured flag: claims_all_backed_by_worker_commits="
        f"{str(evidence.get('claims_all_backed_by_worker_commits')).lower()}"
    )
    lines.append(
        "`uncommitted_only`, `recovery_commit_only` and `absent` are PROMISES, not delivered "
        "work: a recovery commit is harness-created wreckage no Worker self-checked, and an "
        "uncommitted file is not on the branch. Only `worker_commit` is delivered by this round."
    )
    return "\n".join(lines)
