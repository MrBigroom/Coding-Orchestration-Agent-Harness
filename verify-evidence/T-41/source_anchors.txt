  39  -- Atomic compare-and-swap push. expected_version is the last version observed by the
  40  -- device (0 for a new record). A concurrent remote edit returns a conflict instead of
  41  -- being overwritten.
  42  create or replace function public.orchestrator_sync_push(
  43      p_entity_type text,
  44      p_entity_id uuid,
  45      p_expected_version bigint,
  46      p_version bigint,
  47      p_payload jsonb,
  48      p_origin_device text,
  49      p_device_modified_at timestamptz
  50  )
  51  returns jsonb
  52  language plpgsql
  53  security invoker
  54  set search_path = public, pg_temp
  55  as $$
  56  declare
  57      result public.orchestrator_sync_records%ROWTYPE;
  58  begin
  59      if auth.uid() is null then
  60          raise exception 'authentication required';
  61      end if;
  62      if p_expected_version < 0 or p_version != p_expected_version + 1 then
  63          raise exception 'invalid version transition: % -> %', p_expected_version, p_version;
  64      end if;
  65  
  66      insert into public.orchestrator_sync_records (
  67          user_id, entity_type, entity_id, version, payload,
  68          origin_device, device_modified_at
  69      ) values (
  70          auth.uid(), p_entity_type, p_entity_id, p_version, p_payload,
  71          p_origin_device, p_device_modified_at
  72      )
  73      on conflict (user_id, entity_type, entity_id) do nothing
  74      returning * into result;
  75  
  76      if found then
  77          return jsonb_build_object('status', 'ok', 'record', to_jsonb(result));
  78      end if;
  79  
  80      update public.orchestrator_sync_records
  81      set version = p_version,
  82          payload = p_payload,
  83          origin_device = p_origin_device,
  84          device_modified_at = p_device_modified_at,
  85          server_updated_at = now()
  86      where user_id = auth.uid()
  87        and entity_type = p_entity_type
  88        and entity_id = p_entity_id
  89        and version = p_expected_version
  90      returning * into result;
