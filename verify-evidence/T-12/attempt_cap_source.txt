    68	def _run(session, task, tags=None):
    69	    if tags is not None:
    70	        task = task_service.get(session, task.id)
    71	        task.tags = list(tags)
    72	        session.add(task)
    73	        session.commit()
    74	        session.refresh(task)
    75	    outcome = asyncio.run(run_service.run_task_now(session, task))
    76	    session.refresh(task)
    77	    return outcome

    84	def _last_task_status(session, task, needle):
    85	    for entry in log_service.recent(session, task_id=task.id, limit=500):
    86	        if entry.event == "task_status" and needle in entry.details:
    87	            return entry
    88	    raise AssertionError(f"no task_status event containing {needle!r}")

    91	def _cancel_soon(task_id, delay=0.3):
    92	    def cancel():
    93	        time.sleep(delay)
    94	        from orchestrator.core.db import session_scope
    95	
    96	        with session_scope() as s:
    97	            run_service.cancel(s, task_service.get(s, task_id))
    98	
    99	    threading.Thread(target=cancel).start()
